self.addEventListener('install', (event) => {
  event.waitUntil((async () => {
    const cache = await caches.open(CACHE_VERSION);
    await Promise.all(APP_SHELL.map(async (url) => {
      try {
        await cache.add(new Request(url, { cache: 'reload' }));
      } catch (error) {
        console.warn('[FORGE SW] Could not precache', url, error);
      }
    }));
  })());
});

self.addEventListener('activate', (event) => {
  event.waitUntil((async () => {
    const keys = await caches.keys();
    await Promise.all(keys.filter((key) => key !== CACHE_VERSION).map((key) => caches.delete(key)));
    await self.clients.claim();
  })());
});

self.addEventListener('message', (event) => {
  if (event.data && event.data.type === 'SKIP_WAITING') {
    self.skipWaiting();
  }
});

self.addEventListener('fetch', (event) => {
  const request = event.request;
  if (request.method !== 'GET') return;
  const url = new URL(request.url);
  if (url.origin !== self.location.origin) return;
  if (url.pathname.startsWith('/api/')) return;

  if (request.mode === 'navigate') {
    event.respondWith((async () => {
      try {
        return await fetch(request);
      } catch {
        const cache = await caches.open(CACHE_VERSION);
        return (await cache.match('./index.html')) || Response.error();
      }
    })());
    return;
  }

  event.respondWith((async () => {
    const cache = await caches.open(CACHE_VERSION);
    const cached = await cache.match(request);
    if (cached) return cached;
    try {
      const response = await fetch(request);
      if (response.ok) cache.put(request, response.clone());
      return response;
    } catch {
      return cached || Response.error();
    }
  })());
});

  'html','htm','css','js','mjs','cjs','jsx','ts','tsx','json','md','txt',

export function uuid() {
  if (crypto.randomUUID) return crypto.randomUUID();
  const bytes = crypto.getRandomValues(new Uint8Array(16));
  bytes[6] = (bytes[6] & 0x0f) | 0x40;
  bytes[8] = (bytes[8] & 0x3f) | 0x80;
  const hex = [...bytes].map((b) => b.toString(16).padStart(2, '0')).join('');
  return `${hex.slice(0,8)}-${hex.slice(8,12)}-${hex.slice(12,16)}-${hex.slice(16,20)}-${hex.slice(20)}`;
}

export function normalizePath(input) {
  let path = String(input ?? '').replace(/\\/g, '/').trim();
  path = path.replace(/^\/+/, '').replace(/\/+$/, '');
  if (!path) throw new Error('مسیر نمی‌تواند خالی باشد.');
  if (path.includes('\0')) throw new Error('مسیر شامل نویسهٔ غیرمجاز است.');
  if (path.length > 240) throw new Error('مسیر بیش از حد طولانی است.');
  const parts = path.split('/');
  for (const part of parts) {
    if (!part || part === '.' || part === '..') {
      throw new Error('مسیر باید نسبی و بدون بخش‌های . یا .. باشد.');
    }
    if (/[<>:"|?*]/.test(part)) {
      throw new Error('مسیر شامل نویسهٔ غیرمجاز ویندوز است.');
    }
  }
  return parts.join('/');
}

export function extname(path) {
  const name = path.split('/').pop() || '';
  const index = name.lastIndexOf('.');
  return index > 0 ? name.slice(index + 1).toLowerCase() : '';
}

export function basename(path) {
  return path.split('/').pop() || path;
}

export function dirname(path) {
  const index = path.lastIndexOf('/');
  return index === -1 ? '' : path.slice(0, index);
}

export function isTextPath(path) {
  return TEXT_EXTENSIONS.has(extname(path));
}

export function mimeFor(path) {
  const ext = extname(path);
  const map = {
    html:'text/html', htm:'text/html', css:'text/css', js:'text/javascript',
    mjs:'text/javascript', json:'application/json', svg:'image/svg+xml',
    png:'image/png', jpg:'image/jpeg', jpeg:'image/jpeg', gif:'image/gif',
    webp:'image/webp', avif:'image/avif', ico:'image/x-icon',
    woff:'font/woff', woff2:'font/woff2', ttf:'font/ttf', otf:'font/otf',
    mp3:'audio/mpeg', mp4:'video/mp4', webm:'video/webm',
    wasm:'application/wasm', xml:'application/xml', txt:'text/plain',
    md:'text/markdown', pdf:'application/pdf'
  };
  return map[ext] || 'application/octet-stream';
}

export function debounce(fn, wait = 300) {
  let timer = null;
  const debounced = (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), wait);
  };
  debounced.flush = (...args) => {
    clearTimeout(timer);
    fn(...args);
  };
  debounced.cancel = () => clearTimeout(timer);
  return debounced;
}

export function formatBytes(bytes) {
  if (!Number.isFinite(bytes) || bytes <= 0) return '0 B';
  const units = ['B','KB','MB','GB'];
  let value = bytes;
  let index = 0;
  while (value >= 1024 && index < units.length - 1) {
    value /= 1024;
    index += 1;
  }
  return `${value.toFixed(index === 0 ? 0 : 1)} ${units[index]}`;
}

export function downloadBlob(blob, filename) {
  const url = URL.createObjectURL(blob);
  const link = document.createElement('a');
  link.href = url;
  link.download = filename;
  link.rel = 'noopener';
  document.body.append(link);
  link.click();
  link.remove();
  setTimeout(() => URL.revokeObjectURL(url), 30000);
}

export function bytesToBase64(bytes) {
  let binary = '';
  const chunk = 0x8000;
  for (let i = 0; i < bytes.length; i += chunk) {
    binary += String.fromCharCode(...bytes.subarray(i, i + chunk));
  }
  return btoa(binary);
}

export function dataUrlFromFile(file) {
  const bytes = file.binary
    ? new Uint8Array(file.content)
    : new TextEncoder().encode(file.content);
  return `data:${file.mime || mimeFor(file.path)};base64,${bytesToBase64(bytes)}`;
}

export class StorageError extends Error {
  constructor(message, cause) {
    super(message);

export class ForgeDB {
  constructor() {
    this.db = null;
    this._opening = null;
  }

  async open() {
    if (this.db) return this.db;
    if (!this._opening) {
      this._opening = this._openInternal().catch((error) => {
        this._opening = null;
        throw new StorageError('باز کردن پایگاه‌داده ممکن نشد. آیا حالت خصوصی Safari فعال است؟', error);
      });
    }
    return this._opening;
  }

  _openInternal() {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open(DB_NAME, DB_VERSION);

      request.onupgradeneeded = (event) => {
        const db = request.result;
        const tx = request.transaction;
        const oldVersion = event.oldVersion;

        if (!db.objectStoreNames.contains(STORES.projects)) {
          const store = db.createObjectStore(STORES.projects, { keyPath: 'id' });
          store.createIndex('name', 'name');
          store.createIndex('updatedAt', 'updatedAt');
        }

        if (!db.objectStoreNames.contains(STORES.files)) {
          const store = db.createObjectStore(STORES.files, { keyPath: ['projectId', 'path'] });
          store.createIndex('projectId', 'projectId');
          store.createIndex('projectId_updatedAt', ['projectId', 'updatedAt']);
        } else if (oldVersion < 3) {
          const store = tx.objectStore(STORES.files);
          if (!store.indexNames.contains('projectId_updatedAt')) {
            store.createIndex('projectId_updatedAt', ['projectId', 'updatedAt']);
          }
          store.openCursor().onsuccess = (cursorEvent) => {
            const cursor = cursorEvent.target.result;
            if (!cursor) return;
            const value = cursor.value;
            if (!value.kind) {
              cursor.update({
                ...value,
                kind: 'file',
                size: value.size ?? (typeof value.content === 'string'
                  ? value.content.length
                  : value.content?.byteLength || 0)
              });
            }
            cursor.continue();
          };
        }

        if (!db.objectStoreNames.contains(STORES.workspace)) {
          db.createObjectStore(STORES.workspace, { keyPath: 'id' });
        }

        if (!db.objectStoreNames.contains(STORES.snapshots)) {
          const store = db.createObjectStore(STORES.snapshots, { keyPath: 'id' });
          store.createIndex('projectId', 'projectId');
          store.createIndex('projectId_createdAt', ['projectId', 'createdAt']);
        }

        if (!db.objectStoreNames.contains(STORES.syncState)) {
          db.createObjectStore(STORES.syncState, { keyPath: 'projectId' });
        }

        if (!db.objectStoreNames.contains(STORES.prefs)) {
          db.createObjectStore(STORES.prefs, { keyPath: 'key' });
        }
      };

      request.onsuccess = () => {
        this.db = request.result;
        this.db.onversionchange = () => {
          this.db.close();
          this.db = null;
          this._opening = null;
        };
        resolve(this.db);
      };

      request.onerror = () => reject(request.error);
      request.onblocked = () => reject(new Error('پایگاه‌داده توسط نسخهٔ دیگری از برنامه باز نگه داشته شده است.'));
    });

    if (source === target) return;

      throw new Error('انتقال یک پوشه به داخل خودش مجاز نیست.');
    }

    const db = await this.open();
    return new Promise((resolve, reject) => {
      const tx = db.transaction(STORES.files, 'readwrite');
      tx.oncomplete = () => resolve();
      tx.onerror = () => reject(tx.error);
      tx.onabort = () => reject(tx.error || new Error('تغییر نام لغو شد.'));

      const store = tx.objectStore(STORES.files);
      const request = store.index('projectId').getAll(IDBKeyRange.only(projectId));
      request.onsuccess = () => {
        const all = request.result;
        const moving = all.filter(
          (file) => file.path === source || file.path.startsWith(`${source}/`)
        );
        if (!moving.length) {
          reject(new Error('مسیر مبدأ پیدا نشد.'));
          tx.abort();
          return;
        }

        const movingPaths = new Set(moving.map((file) => file.path));
        const updates = [];

        for (const file of moving) {
          const suffix = file.path.slice(source.length);
          const nextPath = normalizePath(target + suffix);
          const conflict = all.find((other) => other.path === nextPath && !movingPaths.has(other.path));
          if (conflict) {
            reject(new Error(`مسیر مقصد «${nextPath}» از قبل وجود دارد.`));
            tx.abort();
            return;
          }
          updates.push({ ...file, path: nextPath, updatedAt: Date.now() });
        }

        for (const file of moving) store.delete([projectId, file.path]);
        for (const file of updates) store.put(file);
      };
    });
  }

function languageFor(CM, path) {
  const ext = path.split('.').pop()?.toLowerCase();
  if (['html', 'htm', 'vue', 'svelte'].includes(ext)) return CM.html();
  if (ext === 'css') return CM.css();
  if (['js', 'mjs', 'cjs', 'jsx', 'ts', 'tsx', 'json'].includes(ext)) {
    return CM.javascript({ jsx: ext === 'jsx', typescript: ['ts','tsx'].includes(ext) });
  }
  return [];
}

export async function createEditor({ parent, prefs, onChange, onFocusChange }) {

      states.set(path, makeState(path, content));
    }
    if (activePath === path && view.state === states.get(path)) {
      view.focus();
      return;
    }
    if (activePath && view.state) {
      states.set(activePath, view.state);
    }
    switching = true;
    activePath = path;
    view.setState(states.get(path));
    switching = false;
    view.focus();
  }

  function invalidate(path) {
    if (activePath === path) return;

      scrollIntoView: true
    });
    view.focus();

    else if (command === 'right') cursorCharRight(view);
    else if (command === 'up') cursorLineUp(view);
    else if (command === 'down') cursorLineDown(view);

      selection: EditorSelection.cursor(line.from),
      scrollIntoView: true
    });
    view.focus();

      fontSizeCompartment.reconfigure(EditorView.theme({

function resolvePath(basePath, reference) {
  const clean = String(reference || '').split('#')[0].split('?')[0].trim();
  if (!clean || /^(?:[a-z]+:)?\/\//i.test(clean) || clean.startsWith('data:') || clean.startsWith('blob:')) {
    return null;
  }
  let path = clean.replace(/\\/g, '/');
  if (path.startsWith('/')) {
    path = path.slice(1);
  } else {
    const base = dirname(basePath);
    path = base ? `${base}/${path}` : path;
  }
  const parts = [];
  for (const part of path.split('/')) {
    if (!part || part === '.') continue;
    if (part === '..') parts.pop();
    else parts.push(part);
  }
  return parts.join('/');
}

function cssUrlToDataUrl(css, cssPath, fileMap, warnings) {
  return css.replace(/url\(\s*(['"]?)(.*?)\1\s*\)/gi, (match, quote, reference) => {
    if (/^(data:|https?:|blob:|#)/i.test(reference)) return match;
    const resolved = resolvePath(cssPath, reference);
    const file = resolved ? fileMap.get(resolved) : null;
    if (!file) {
      warnings.push(`CSS asset not found: ${reference} (from ${cssPath})`);
      return match;
    }
    return `url("${dataUrlFromFile(file)}")`;
  });
}

function buildPreviewHtml(project, files) {
  const fileMap = new Map(files.map((file) => [file.path, file]));
  const htmlFile = fileMap.get('index.html')
    || files.find((file) => file.kind === 'file' && file.path.toLowerCase() === 'index.html');

  const warnings = [];
  if (!htmlFile) {
    return {
      html: `
        
index.html not found

        
Create an index.html at the project root to preview.
`,
      scripts: [],
      warnings: ['index.html not found']
    };
  }

  const doc = parseHtmlDocument(htmlFile.content);
  doc.querySelectorAll('meta[http-equiv="Content-Security-Policy"]').forEach((node) => node.remove());
  doc.querySelectorAll('base').forEach((node) => node.remove());

  const csp = doc.createElement('meta');
  csp.setAttribute('http-equiv', 'Content-Security-Policy');
  csp.setAttribute('content', [
    "default-src 'none'",
    "script-src blob:",
    "style-src 'unsafe-inline'",
    "img-src data: blob:",
    "font-src data: blob:",
    "media-src data: blob:",
    "connect-src 'none'",
    "base-uri 'none'",
    "form-action 'none'",
    "frame-src 'none'",
    "object-src 'none'"
  ].join('; '));
  doc.head.prepend(csp);

  const scripts = [];

  doc.querySelectorAll('link[rel~="stylesheet"]').forEach((link) => {
    const href = link.getAttribute('href');
    if (!href) { link.remove(); return; }
    const resolved = resolvePath(htmlFile.path, href);
    const cssFile = resolved ? fileMap.get(resolved) : null;
    if (!cssFile || cssFile.binary) {
      warnings.push(`Stylesheet not resolved locally: ${href}`);
      link.remove();
      return;
    }
    const style = doc.createElement('style');
    style.textContent = cssUrlToDataUrl(cssFile.content, cssFile.path, fileMap, warnings);
    link.replaceWith(style);
  });

  doc.querySelectorAll('script').forEach((script) => {
    const src = script.getAttribute('src');
    if (src) {
      const resolved = resolvePath(htmlFile.path, src);
      const jsFile = resolved ? fileMap.get(resolved) : null;
      if (!jsFile || jsFile.binary) {
        warnings.push(`Script not resolved locally: ${src}`);
        script.remove();
        return;
      }
      scripts.push({
        code: jsFile.content,
        type: script.getAttribute('type') === 'module' ? 'module' : 'classic'
      });
    } else {
      scripts.push({
        code: script.textContent || '',
        type: script.getAttribute('type') === 'module' ? 'module' : 'classic'
      });
    }
    script.remove();
  });

  const assetAttributes = ['src', 'poster'];
  doc.querySelectorAll('img, source, video, audio, track').forEach((element) => {
    for (const attribute of assetAttributes) {
      const value = element.getAttribute(attribute);
      if (!value || /^(data:|https?:|blob:)/i.test(value)) continue;
      const resolved = resolvePath(htmlFile.path, value);
      const asset = resolved ? fileMap.get(resolved) : null;
      if (!asset) {
        warnings.push(`Asset not found: ${value}`);
        continue;
      }
      element.setAttribute(attribute, dataUrlFromFile(asset));
    }
  });

  doc.querySelectorAll('iframe, object, embed').forEach((node) => {
    warnings.push(`Embedded element removed: <${node.tagName.toLowerCase()}>`);
    node.remove();
  });

  const bootstrap = doc.createElement('script');
  bootstrap.src = URL.createObjectURL(new Blob([BOOTSTRAP_SOURCE], { type: 'text/javascript' }));
  bootstrap.dataset.forgeBootstrap = 'true';
  doc.body.append(bootstrap);

  return { html: serializeDocument(doc), scripts, warnings, bootstrapUrl: bootstrap.src };
}

export class Preview {
  constructor({ host, onLog, onError }) {
    this.host = host;
    this.onLog = onLog;
    this.onError = onError;
    this.iframe = null;
    this.bootstrapUrl = null;
    this.scripts = [];
    this._messageHandler = null;
  }

  async run(project, files) {
    this.destroyFrame();
    const built = buildPreviewHtml(project, files);
    for (const warning of built.warnings) this.onLog?.({ level: 'warn', args: [warning] });

    const iframe = document.createElement('iframe');
    iframe.setAttribute('sandbox', 'allow-scripts');
    iframe.setAttribute('referrerpolicy', 'no-referrer');
    iframe.setAttribute('title', 'Project preview');
    iframe.srcdoc = built.html;
    this.iframe = iframe;
    this.bootstrapUrl = built.bootstrapUrl;
    this.scripts = built.scripts;

    this._messageHandler = (event) => {
      if (!this.iframe || event.source !== this.iframe.contentWindow) return;
      const data = event.data;
      if (!data || data.__forge !== true) return;
      if (data.type === 'ready') {
        try {
          this.iframe.contentWindow.postMessage({
            __forge: true,
            type: 'run',
            payload: { scripts: this.scripts.slice(0, 40) }
          }, '*');
        } catch (error) {
          this.onError?.(error);
        }
      } else if (data.type === 'log') {
        const payload = data.payload || {};
        this.onLog?.({
          level: ['log','info','warn','error'].includes(payload.level) ? payload.level : 'log',
          args: Array.isArray(payload.args) ? payload.args.slice(0, 10) : []
        });
      } else if (data.type === 'run-accepted') {
        this.onLog?.({ level: 'info', args: [`Preview executed ${data.payload?.count ?? 0} script(s).`] });
      }
    };
    window.addEventListener('message', this._messageHandler);

    this.host.replaceChildren(iframe);
    return built.warnings;
  }

  destroyFrame() {
    if (this._messageHandler) {
      window.removeEventListener('message', this._messageHandler);
      this._messageHandler = null;
    }
    if (this.iframe) {
      this.iframe.remove();
      this.iframe = null;
    }
    if (this.bootstrapUrl) {
      URL.revokeObjectURL(this.bootstrapUrl);
      this.bootstrapUrl = null;
    }
  }

  destroy() {
    this.destroyFrame();
  }
}

const LIMITS = {
  maxEntries: 500,
  maxFileBytes: 8 * 1024 * 1024,
  maxTotalBytes: 30 * 1024 * 1024
};

function unzipAsync(data) {
  return new Promise((resolve, reject) => {
    unzip(data, (error, result) => {
      if (error) reject(error);
      else resolve(result);
    });
  });
}

export async function readZip(file) {
  if (!file) throw new Error('فایلی انتخاب نشد.');
  if (file.size > LIMITS.maxTotalBytes * 2) {
    throw new Error('حجم فایل ZIP بیش از حد مجاز است.');
  }
  const bytes = new Uint8Array(await file.arrayBuffer());
  let archive;
  try {
    archive = await unzipAsync(bytes);
  } catch (error) {
    throw new Error('ZIP معتبر نیست یا فشرده‌سازی آن پشتیبانی نمی‌شود.', { cause: error });
  }

  const paths = Object.keys(archive);
  if (paths.length > LIMITS.maxEntries) {
    throw new Error(`تعداد ورودی‌های ZIP بیش از ${LIMITS.maxEntries} است.`);
  }

  const entries = [];
  const seen = new Set();
  let total = 0;

  for (const rawPath of paths) {
    const isDirectory = rawPath.endsWith('/');
    let path;
    try {
      path = normalizePath(isDirectory ? rawPath.slice(0, -1) : rawPath);
    } catch (error) {
      throw new Error(`مسیر ناامن در ZIP: ${rawPath}`);
    }

    if (path.startsWith('__MACOSX/') || path.endsWith('.DS_Store')) continue;
    if (seen.has(path)) throw new Error(`مسیر تکراری در ZIP: ${path}`);
    seen.add(path);

    const content = archive[rawPath];
    const size = content.byteLength;
    total += size;
    if (size > LIMITS.maxFileBytes) throw new Error(`فایل «${path}» بیش از حد بزرگ است.`);
    if (total > LIMITS.maxTotalBytes) throw new Error('حجم استخراج‌شدهٔ ZIP بیش از حد مجاز است.');

    if (isDirectory) {
      entries.push({ path, kind: 'folder', content: '', binary: false, mime: '', size: 0 });
      continue;
    }

    const text = isTextPath(path);
    entries.push({
      path,
      kind: 'file',
      content: text ? strFromU8(content) : content.buffer.slice(content.byteOffset, content.byteOffset + content.byteLength),
      binary: !text,
      mime: mimeFor(path),
      size
    });
  }

  const files = entries.filter((entry) => entry.kind === 'file');
  if (!files.length) throw new Error('ZIP هیچ فایل قابل‌استفاده‌ای ندارد.');
  return entries;
}

function buildZipInput(files) {
  const input = {};
  for (const file of files) {
    if (file.kind === 'folder') {
      input[`${file.path}/`] = new Uint8Array(0);
    } else if (file.binary) {
      input[file.path] = new Uint8Array(file.content);
    } else {
      input[file.path] = strToU8(file.content);
    }
  }
  return input;
}

function zipAsync(input) {
  return new Promise((resolve, reject) => {
    zip(input, { level: 6 }, (error, data) => {
      if (error) reject(error);
      else resolve(data);
    });
  });
}

export async function exportProjectZip(project, files) {
  const input = buildZipInput(files);
  const zipped = await zipAsync(input);
  if (!zipped || !zipped.byteLength) throw new Error('ساخت ZIP ناموفق بود.');

  const verification = unzipSync(zipped);
  const expected = Object.keys(input).sort();
  const actual = Object.keys(verification).sort();
  if (expected.length !== actual.length || expected.some((key, index) => key !== actual[index])) {
    throw new Error('صحت ZIP ساخته‌شده تأیید نشد.');
  }

  const filename = `${sanitizeFilename(project.name)}.zip`;
  downloadBlob(new Blob([zipped], { type: 'application/zip' }), filename);
  return { filename, bytes: zipped.byteLength };
}

export function searchProject(files, query, { caseSensitive = false, maxResults = 200 } = {}) {
  const needle = String(query || '');
  if (!needle) return { results: [], truncated: false };
  const haystackQuery = caseSensitive ? needle : needle.toLowerCase();
  const results = [];
  let truncated = false;

  for (const file of files) {
    if (file.kind !== 'file' || file.binary || !isTextPath(file.path)) continue;
    const lines = String(file.content || '').split('\n');
    for (let lineIndex = 0; lineIndex < lines.length; lineIndex += 1) {
      const line = lines[lineIndex];
      const haystack = caseSensitive ? line : line.toLowerCase();
      let column = haystack.indexOf(haystackQuery);
      while (column !== -1) {
        results.push({
          path: file.path,
          line: lineIndex + 1,
          column: column + 1,
          preview: line.trim().slice(0, 220)
        });
        if (results.length >= maxResults) {
          truncated = true;
          return { results, truncated };
        }
        column = haystack.indexOf(haystackQuery, column + Math.max(1, haystackQuery.length));
      }
    }
  }
  return { results, truncated };
}

export async function replaceAll(db, projectId, files, query, replacement, { caseSensitive = false } = {}) {
  const needle = String(query || '');
  if (!needle) throw new Error('متن جستجو خالی است.');

  let count = 0;
  const updated = [];

  for (const file of files) {
    if (file.kind !== 'file' || file.binary || !isTextPath(file.path)) continue;
    const source = String(file.content || '');
    let next;
    if (caseSensitive) {
      next = source.split(needle).join(replacement);
      count += source.split(needle).length - 1;
    } else {
      const escaped = needle.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
      const regex = new RegExp(escaped, 'gi');
      const matches = source.match(regex);
      count += matches ? matches.length : 0;
      next = source.replace(regex, replacement);
    }
    if (next !== source) {
      updated.push({
        ...file,
        content: next,
        size: next.length,
        updatedAt: Date.now()
      });
    }
  }

  if (updated.length) {
    await db.putFilesBatch(updated);
  }
  return { count, filesChanged: updated.length };
}

export function el(tag, attrs = {}, ...children) {
  const node = document.createElement(tag);
  for (const [key, value] of Object.entries(attrs || {})) {
    if (key === 'class') node.className = value;
    else if (key === 'text') node.textContent = value;
    else if (key === 'dataset') Object.assign(node.dataset, value);
    else if (key.startsWith('on') && typeof value === 'function') {
      node.addEventListener(key.slice(2).toLowerCase(), value);
    } else if (value !== undefined && value !== null && value !== false) {
      node.setAttribute(key, value === true ? '' : String(value));
    }
  }
  for (const child of children.flat(Infinity)) {
    if (child === null || child === undefined || child === false) continue;
    node.append(child.nodeType ? child : document.createTextNode(String(child)));
  }
  return node;
}

function modal({ title, message, input = null, actions = [] }) {
  return new Promise((resolve) => {
    const root = document.getElementById('modal-root');
    const backdrop = el('div', { class: 'modal-backdrop' });
    const dialog = el('div', { class: 'modal', role: 'dialog', 'aria-modal': 'true' });
    dialog.append(el('h3', { text: title }));
    if (message) dialog.append(el('p', { text: message }));

    let inputNode = null;
    if (input) {
      inputNode = el('input', {
        type: input.type || 'text',
        value: input.value || '',
        placeholder: input.placeholder || '',
        autocomplete: 'off',
        spellcheck: 'false'
      });
      dialog.append(inputNode);
    }

    const actionRow = el('div', { class: 'modal-actions' });
    let settled = false;
    const finish = (value) => {
      if (settled) return;
      settled = true;
      backdrop.remove();
      resolve(value);
    };

    for (const action of actions) {
      actionRow.append(el('button', {
        type: 'button',
        text: action.label,
        dataset: {
          primary: action.primary ? 'true' : 'false',
          danger: action.danger ? 'true' : 'false'
        },
        onclick: () => finish(action.value(inputNode))
      }));
    }

    dialog.append(actionRow);
    backdrop.append(dialog);
    backdrop.addEventListener('click', (event) => {
      if (event.target === backdrop) finish(null);
    });
    root.append(backdrop);
    inputNode?.focus();
    inputNode?.select();
  });
}

function renderFileTree(container) {
  container.replaceChildren();
  const tree = new Map();

  for (const entry of workspace.files) {
    const parts = entry.path.split('/');
    let level = tree;
    for (let index = 0; index < parts.length; index += 1) {
      const name = parts[index];
      const path = parts.slice(0, index + 1).join('/');
      const isLeaf = index === parts.length - 1;
      if (!level.has(name)) {
        level.set(name, {
          name, path,
          kind: isLeaf ? entry.kind : 'folder',
          entry: isLeaf ? entry : null,
          children: new Map()
        });
      }
      const node = level.get(name);
      if (isLeaf && entry.kind === 'folder') node.kind = 'folder';
      if (!isLeaf) level = node.children;
    }
  }

  const renderLevel = (map, depth, target) => {
    const sorted = [...map.values()].sort((a, b) => {
      if (a.kind !== b.kind) return a.kind === 'folder' ? -1 : 1;
      return a.name.localeCompare(b.name, 'fa');
    });

    for (const node of sorted) {
      const isFolder = node.kind === 'folder';
      const expanded = workspace.expandedFolders.has(node.path);
      const row = el('div', { class: 'tree-row' });
      const main = el('button', {
        type: 'button',
        class: 'tree-main',
        style: `padding-inline-start:${depth * 12}px`,
        'aria-label': isFolder ? `${expanded ? 'بستن' : 'باز کردن'} پوشهٔ ${node.path}` : `باز کردن ${node.path}`,
        onclick: async () => {
          if (isFolder) {
            if (expanded) workspace.expandedFolders.delete(node.path);
            else workspace.expandedFolders.add(node.path);
            renderFileTree(container);
          } else {
            await workspace.openFile(node.path);
            closeSheet();
          }
        }
      });
      main.append(el('span', { class: 'tree-caret', text: isFolder ? (expanded ? '▾' : '▸') : '·' }));
      main.append(el('span', { class: 'name', text: node.name }));
      row.append(main);

      if (!isFolder && node.entry) {
        row.append(el('button', {
          type: 'button',
          class: 'tree-action',
          'aria-label': `عملیات ${node.path}`,
          text: '⋯',
          onclick: () => openEntryActions(node.entry)
        }));
      }

      target.append(row);

      if (isFolder && expanded) {
        renderLevel(node.children, depth + 1, target);
      }
    }
  };

  const wrapper = el('div', { class: 'tree' });
  renderLevel(tree, 0, wrapper);
  container.append(wrapper);
}

function setupToolbar() {
  dom.toolbar.addEventListener('click', async (event) => {
    const button = event.target.closest('button[data-cmd]');

    if (!editor) {
      toast('ویرایشگر آماده نیست. ابتدا vendor را بسازید.', { type: 'error' });

    if (command === 'save') {
      try {
        await workspace.saveAll();
        toast('همهٔ تغییرات ذخیره شد.');
      } catch (error) {
        toast(`ذخیره ناموفق: ${error.message}`, { type: 'error' });
      }
      return;
    }
    if (command === 'pair') {
      editor.pair(button.dataset.open || '', button.dataset.close || '');
      return;
    }
    if (command === 'semicolon') {
      editor.semicolon();
      return;
    }
    if (command === 'indent') editor.indent();
    else if (command === 'outdent') editor.outdent();
    else if (command === 'undo') editor.undo();
    else if (command === 'redo') editor.redo();
    else if (command === 'left' || command === 'right' || command === 'up' || command === 'down') editor.move(command);
    else if (command === 'search') editor.search();
  });
}

async function registerServiceWorker() {
  if (!('serviceWorker' in navigator)) return;
  try {
    const registration = await navigator.serviceWorker.register('./sw.js', { scope: './', updateViaCache: 'none' });
    registration.addEventListener('updatefound', () => {
      const worker = registration.installing;
      if (!worker) return;
      worker.addEventListener('statechange', () => {
        if (worker.state === 'installed' && navigator.serviceWorker.controller) {
          toast('نسخهٔ جدید FORGE آماده است.', {
            duration: 12000,
            action: {
              label: 'به‌روزرسانی',
              onClick: () => {
                worker.postMessage({ type: 'SKIP_WAITING' });
                navigator.serviceWorker.addEventListener('controllerchange', () => location.reload(), { once: true });
              }
            }
          });
        }
      });
    });
  } catch (error) {
    console.warn('[FORGE] service worker registration failed', error);
    toast('ثبت Service Worker ناموفق بود؛ برنامه ممکن است آفلاین کار نکند.', { type: 'warn' });
  }
}

# Architecture — NONONICK FORGE Milestone 1

## Runtime model
Static SPA. No backend is required for local editing, storage, preview, or ZIP.
The browser is the runtime. A one-time development build bundles pinned dependencies
into `vendor/`; normal app usage does not require a build.

## Modules
- `storage.js`: versioned IndexedDB schema, transactions, migrations, snapshots.
- `workspace.js`: in-memory project state, dirty tracking, autosave, session restore.
- `editor.js`: CodeMirror 6 integration, per-file EditorState, toolbar commands.
- `preview.js`: builds a sandboxed preview document, resolves local assets to data URLs,
  extracts scripts, runs them through a blob bootstrap, and captures errors/logs.
- `archive.js`: fflate ZIP import/export with path and size validation.
- `search.js`: plain-text project search and transaction-safe replace-all.
- `ui.js`: accessible modals, toasts, DOM helpers.
- `app.js`: wiring, routing, sheets, toolbar, PWA registration.
- `sw.js`: versioned app-shell cache; never caches project data or API responses.

## Storage schema
Database: `nononick-forge`, version 3.

| Store | Key | Purpose |
|---|---|---|
| `projects` | `id` | Project metadata |
| `files` | `[projectId, path]` | Files and folder records |
| `workspace` | `id` | Last session |
| `snapshots` | `id` | Local revisions |
| `syncState` | `projectId` | Reserved for future sync metadata |
| `prefs` | `key` | Non-sensitive preferences |

Folders are explicit records with `kind: "folder"`. Empty folders can be persisted.
Binary files are stored as `ArrayBuffer`; text files as strings.

## Preview isolation
- iframe with `sandbox="allow-scripts"` only.
- No `allow-same-origin`, no top navigation, no popups, no forms.
- Project scripts are delivered as blob URLs created inside the sandboxed document
  by a bootstrap blob script.
- Project CSS is inlined after local `url(...)` assets are converted to data URLs.
- The preview document receives a restrictive CSP.
- Parent and preview communicate only via `postMessage`; the parent validates
  `event.source === iframe.contentWindow` and known message types.
- No app secrets or privileged APIs are exposed to preview code.

## PWA
- `manifest.webmanifest` with standalone display.
- Versioned service worker caches the app shell.
- Project data remains in IndexedDB; the service worker never caches it.
- Offline behavior is best-effort and subject to Safari/WebKit storage and
  service-worker limitations.

# Security and Dependency Report

## Main application
- No `eval()` or `new Function()` in application code.
- CSP: no `script-src 'unsafe-inline'` and no `script-src 'unsafe-eval'`.
- `script-src 'self' blob:` is required because the sandboxed preview runs project
  scripts from blob URLs. Application code itself is external modules.
- `style-src 'self' 'unsafe-inline'` is required by CodeMirror style injection and
  preview inline CSS.
- `connect-src` is limited to same-origin and `https://api.github.com`. GitHub
  integration is not wired in this milestone.
- All dynamic text in the main UI uses `textContent` or `createElement`, not
  unsanitized `innerHTML`.
- Imported paths are normalized and validated; traversal and absolute paths are rejected.
- ZIP entries are bounded by count, per-file size, and total size.

## Preview
- Opaque-origin sandbox: `sandbox="allow-scripts"` only.
- No `allow-same-origin`.
- Preview code cannot access parent DOM, IndexedDB, localStorage, or app variables.
- Preview network access is restricted by CSP (`connect-src 'none'`); external
  resources are generally blocked.
- Runtime errors are captured and displayed; failure messages are not hidden.

## GitHub
- Not configured. OAuth token endpoints do not provide CORS headers for browser-only
  static apps, and a client secret must not be embedded.
- A future implementation requires a stateless relay with a narrow allow-list and a
  documented API contract. Tokens must not be written to project files, ZIP exports,
  or ordinary persistent browser storage without a justified security design.

## Dependencies
Pinned production dependencies:
- CodeMirror 6 packages: `6.65.7`, `6.20.4`, `6.11.1`, `6.3.1`, `6.4.12`, `6.2.5`,
  `6.13.1`, `6.7.2`, `6.7.6`, `6.43.15`
- fflate `0.8.3`

Development dependencies: esbuild `0.24.2`, sharp `0.33.5`.
Run `npm audit` after installing; commit `package-lock.json`. This report does not
claim a vulnerability-free status for transitive dependencies.

# Limitations and Explicit Non-Features

## GitHub
Not implemented in Milestone 1. OAuth token endpoints lack CORS for browser-only
static apps and a client secret cannot be embedded. A future milestone must define a
stateless relay contract. No fake Git actions are shown.

## Preview
- Node.js, npm packages, shell commands, and backend execution are not supported.
- `fetch` and external network resources are blocked by CSP in the preview.
- Relative ES module imports inside project JavaScript are not recursively resolved.
- Inline event handler attributes may be blocked by CSP; use `addEventListener`.
- External stylesheets/scripts are not loaded; they are reported as warnings.
- Preview behavior is same-origin iframe sandboxed with an opaque origin. It is not
  a substitute for a dedicated preview origin; test against the app’s real CSP before
  production use.

## Editor
- CodeMirror syntax highlighting is not full language-server diagnostics.
- No LSP, linting, or TypeScript type checking.
- Binary files are not editable.

## Storage
- IndexedDB is not permanent. Browser or user site-data deletion removes projects.
- Users should export ZIP or JSON backups regularly.
- Storage quota is browser-managed.

## iOS
- File System Access API is unavailable on iOS Safari.
- PWA installation is manual via Share → Add to Home Screen.
- Service worker and storage behavior depend on WebKit.
- VisualViewport keyboard handling is best-effort; test all orientations.

## Search
- Plain-text search only; no regex UI in this milestone.
